// ═══ Réservations (js/resa.js, reserver/, functions/src/resa.js): the rules, as merged in firestore.rules (reference copy) ═══
// firestore.rules is the source of truth; tests: tools/resa.test.js (emulator), with tools/rules.test.js.
//
// Model:
//   restos/{rid}/public/resa     projection written by the app (any member): { name, address, phone, photo, msg, mail (server live),
//                                cfg:{ services, slot, maxPerSlot, maxService, closures, online, validate, waitlist, lead, maxOnline, horizon },
//                                days:{ "YYYY-MM-DD": { "HH:MM": coversBooked } }, last?, updatedAt }   — no client data in it.
//   restos/{rid}/bookings/{id}   a booking. The public page creates one, without an account, in ONE batch with:
//                                  · status "en ligne" (cfg.validate off) or "avalider" (cfg.validate on): + covers on
//                                    public/resa days[date][time] and public/resa.last = the booking id;
//                                  · status "attente" (waiting list, cfg.waitlist not false): no counter, holds no seat;
//                                  · resaKeys/{rid}/keys/{id} = { k: 32 hex, at } — the private key of the « Modifier ou annuler » link.
//   resaKeys/{rid}/keys/{id}     create-only with its new booking, never readable or writable afterwards by anyone but the server
//                                (Admin SDK: resaManage checks it, onResaBooking creates one for staff bookings that have an e-mail).
//   restos/{rid}/clients/{phone} client file (VIP, notes), members only (generic rule).
// Guarantees: anyone can read the public doc (get only, never list); nobody without an account can read a booking or a key;
// an online booking needs online on, whitelisted fields of bounded size, 1..maxOnline covers, a time in the future past the
// lead time, and a slot that stays within maxPerSlot (checked on the counter written in the same batch).
// Client changes (cancel / modify) never touch Firestore from the browser: they go through the callable resaManage.

// ── helpers (top of firestore.rules) ──
    function resaPubPath(rid) { return /databases/$(database)/documents/restos/$(rid)/public/resa; }
    function resaBookingPath(rid, bid) { return /databases/$(database)/documents/restos/$(rid)/bookings/$(bid); }
    function resaFieldsOk(b) {
      return b.keys().hasOnly(['date', 'time', 'covers', 'name', 'phone', 'email', 'notes', 'optin', 'status', 'source', 'createdAt', 'at'])
        && b.keys().hasAll(['date', 'time', 'covers', 'name', 'phone', 'status', 'source', 'createdAt', 'at'])
        && b.status in ['en ligne', 'avalider', 'attente'] && b.source == 'web'
        && b.date is string && b.date.matches('^20[0-9]{2}-[01][0-9]-[0-3][0-9]$')
        && b.time is string && b.time.matches('^[0-2][0-9]:[0-5][0-9]$')
        && b.covers is int && b.covers >= 1
        && b.name is string && b.name.size() >= 2 && b.name.size() <= 60
        && b.phone is string && b.phone.size() >= 6 && b.phone.size() <= 25
        && (!('email' in b) || (b.email is string && b.email.size() <= 100))
        && (!('notes' in b) || (b.notes is string && b.notes.size() <= 300))
        && (!('optin' in b) || b.optin == true)
        && b.createdAt is int && b.createdAt > request.time.toMillis() - 600000 && b.createdAt < request.time.toMillis() + 600000
        && b.at is int && b.at < request.time.toMillis() + 400 * 86400000;
    }
    function resaOnlineOk(rid, bid) {
      let b = request.resource.data;
      let p = get(resaPubPath(rid)).data;
      let pa = getAfter(resaPubPath(rid)).data;
      let before = p.days.get(b.date, {}).get(b.time, 0);
      return resaFieldsOk(b)
        && p.cfg.online == true
        && b.covers <= p.cfg.maxOnline
        && b.at > request.time.toMillis() + p.cfg.lead * 3600000
        && (b.status == 'attente'
              // waiting list: holds no seat, no counter; only when the restaurant keeps one
              ? p.cfg.get('waitlist', true) != false
              // a booking (confirmed at once, or « à valider » when the restaurant validates each request) takes its seats on the counter
              : b.status == (p.cfg.get('validate', false) == true ? 'avalider' : 'en ligne')
                && pa.last == bid
                && pa.days[b.date][b.time] == before + b.covers
                && (p.cfg.maxPerSlot == 0 || pa.days[b.date][b.time] <= p.cfg.maxPerSlot));
    }
    // the private manage key of a booking (reserver/?r=&b=&k=): written once, with the booking, never readable by a client
    function resaKeyOk(rid, bid) {
      let k = request.resource.data;
      return k.keys().hasOnly(['k', 'at']) && k.k is string && k.k.matches('^[a-f0-9]{32}$')
        && !exists(resaBookingPath(rid, bid)) && existsAfter(resaBookingPath(rid, bid));
    }
    function resaCounterOk(rid) {
      let n = request.resource.data;
      let o = resource.data;
      let bid = n.last;
      let b = getAfter(resaBookingPath(rid, bid)).data;
      return n.diff(o).affectedKeys().hasOnly(['days', 'last'])
        && bid is string
        && !exists(resaBookingPath(rid, bid))
        && existsAfter(resaBookingPath(rid, bid))
        && b.status in ['en ligne', 'avalider']
        && n.days.diff(o.days).affectedKeys().hasOnly([b.date])
        && n.days[b.date].diff(o.days.get(b.date, {})).affectedKeys().hasOnly([b.time])
        && n.days[b.date][b.time] == o.days.get(b.date, {}).get(b.time, 0) + b.covers;
    }

// ── inside match /restos/{rid} ──
      match /public/{doc} {
        allow get: if doc == 'resa';
        // the booking page adds the covers of the booking it creates in the same batch, and nothing else
        allow update: if doc == 'resa' && resaCounterOk(rid);
      }
      // ── réservations: an online booking, created without an account ──
      match /bookings/{bid} {
        allow create: if resaOnlineOk(rid, bid);
      }

// ── top level ──
    match /resaKeys/{rid}/keys/{bid} {
      allow create: if resaKeyOk(rid, bid);
    }
